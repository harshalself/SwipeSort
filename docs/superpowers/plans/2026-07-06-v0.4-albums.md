# SwipeSort v0.4 — Album Sorting Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Prerequisite:** v0.3 tagged. Persistence works.

**Goal:** Extend the swipe gesture with a third decision — swipe UP to sort into a user album. Show a horizontal album picker below the card. Toggle favorite via a heart button. Never mutate smart albums.

**Architecture:** New `SwipeDecision.moveToAlbum(AssetIdentifier, PHAssetCollection.LocalIdentifier)` case. `PhotoLibraryProviding` gains album-listing + album-mutation APIs. Commit step batches deletions and moves in one `PHPhotoLibrary.performChanges`.

**Tech Stack:** Same. Adds `PHAssetCollectionChangeRequest`, `PHAssetChangeRequest.isFavorite`.

## Global Constraints

- All prior constraints apply.
- **Smart albums are read-only.** Every album shown in the picker must return `true` for `canPerform(.addContent)`. Filter aggressively.
- **Move is an actual move**: remove from source user album + add to destination in a single atomic `performChanges` block (per `docs/RESEARCH.md` §2.1). When the swipe originates from All Photos (no source user album), the operation reduces to a plain add.
- **Reviewed status is monotonic**: once an asset is marked reviewed (kept, deleted, or moved), it stays reviewed across future sessions. Moving does NOT re-open the decision. Only undo (session-scoped) clears the reviewed record.
- Favorite toggle is separate from the swipe gesture (a heart button) so it doesn't collide with the three-direction swipe.
- Album creation happens inline via a sheet — no separate management screen in v0.4.

## Research References

- `PHAssetCollectionChangeRequest` semantics + `canPerform` gating: `docs/RESEARCH.md` §2.1.
- Smart album immutability list: §2.5.

```swift
func userAlbums() async -> [UserAlbum] {
    let opts = PHFetchOptions()
    opts.sortDescriptors = [NSSortDescriptor(key: "localizedTitle", ascending: true)]
    let fetched = PHAssetCollection.fetchAssetCollections(
        with: .album, subtype: .albumRegular, options: opts
    )
    var out: [UserAlbum] = []
    fetched.enumerateObjects { col, _, _ in
        let title = col.localizedTitle ?? "Untitled"
        let canAdd = col.canPerform(.addContent)
        let countFetch = PHAsset.fetchAssets(in: col, options: nil)
        out.append(UserAlbum(
            id: col.localIdentifier,
            title: title,
            assetCount: countFetch.count,
            canAdd: canAdd
        ))
    }
    return out
}

func createUserAlbum(title: String) async -> UserAlbum? {
    if DryRunMode.isEnabled {
        DryRunMode.log("Would create album titled \(title)")
        return UserAlbum(id: "dryrun-\(UUID().uuidString)", title: title, assetCount: 0, canAdd: true)
    }
    var placeholderId: String?
    let ok: Bool = await withCheckedContinuation { cont in
        PHPhotoLibrary.shared().performChanges({
            let req = PHAssetCollectionChangeRequest.creationRequestForAssetCollection(withTitle: title)
            placeholderId = req.placeholderForCreatedAssetCollection.localIdentifier
        }, completionHandler: { success, _ in
            cont.resume(returning: success)
        })
    }
    guard ok, let id = placeholderId else { return nil }
    return UserAlbum(id: id, title: title, assetCount: 0, canAdd: true)
}

func moveAssets(_ ids: [AssetIdentifier], fromAlbumId: String?, toAlbumId: String) async -> CommitResult {
    guard !ids.isEmpty else { return .success(affectedCount: 0) }

    if DryRunMode.isEnabled {
        let src = fromAlbumId ?? "AllPhotos"
        DryRunMode.log("Would move \(ids.count) asset(s) from \(src) to \(toAlbumId)")
        return .success(affectedCount: ids.count)
    }

    // Resolve destination — must be writable.
    let destFetch = PHAssetCollection.fetchAssetCollections(withLocalIdentifiers: [toAlbumId], options: nil)
    guard let dest = destFetch.firstObject, dest.canPerform(.addContent) else {
        return .failed(reason: "Destination album not writable.")
    }

    // Resolve source (optional). If provided, it must support .removeContent.
    var source: PHAssetCollection?
    if let fromId = fromAlbumId {
        let srcFetch = PHAssetCollection.fetchAssetCollections(withLocalIdentifiers: [fromId], options: nil)
        guard let s = srcFetch.firstObject, s.canPerform(.removeContent) else {
            return .failed(reason: "Source album does not permit removals (likely a smart album).")
        }
        source = s
    }

    let assets = PHAsset.fetchAssets(withLocalIdentifiers: ids.map(\.rawValue), options: nil)
    return await withCheckedContinuation { cont in
        PHPhotoLibrary.shared().performChanges({
            // Atomic: remove from source + add to destination in one performChanges block.
            if let src = source, let removeReq = PHAssetCollectionChangeRequest(for: src) {
                removeReq.removeAssets(assets as NSFastEnumeration)
            }
            if let addReq = PHAssetCollectionChangeRequest(for: dest) {
                addReq.addAssets(assets as NSFastEnumeration)
            }
        }, completionHandler: { success, error in
            if success {
                cont.resume(returning: .success(affectedCount: ids.count))
            } else if let ns = error as NSError?,
                      ns.domain == PHPhotosErrorDomain,
                      ns.code == PHPhotosError.userCancelled.rawValue {
                cont.resume(returning: .cancelledByUser)
            } else {
                cont.resume(returning: .failed(reason: error?.localizedDescription ?? "unknown"))
            }
        })
    }
}

func setFavorite(_ id: AssetIdentifier, isFavorite: Bool) async -> Bool {
    if DryRunMode.isEnabled {
        DryRunMode.log("Would set favorite=\(isFavorite) on \(id.rawValue)")
        return true
    }
    let assets = PHAsset.fetchAssets(withLocalIdentifiers: [id.rawValue], options: nil)
    guard let asset = assets.firstObject else { return false }
    return await withCheckedContinuation { cont in
        PHPhotoLibrary.shared().performChanges({
            let req = PHAssetChangeRequest(for: asset)
            req.isFavorite = isFavorite
        }, completionHandler: { ok, _ in
            cont.resume(returning: ok)
        })
    }
}
```

Update `commitDecisions()` to also run atomic moves. Group `.moveToAlbum` by the (from, to) pair:

```swift
func commitDecisions() async -> CommitResult {
    let deleteIds = pendingDeletions
    let moves = pendingMoves

    phase = .committing

    // 1) Deletions first. If the user cancels the system delete prompt,
    //    we treat the whole commit as aborted — moves stay pending too so
    //    the user can review the queue and retry, or use undo to change
    //    their mind about specific decisions. This is intentional
    //    all-or-nothing behavior; splitting could leave the user with
    //    "some deletes went through, some didn't" ambiguity.
    var deleteResult: CommitResult = .success(affectedCount: 0)
    if !deleteIds.isEmpty {
        deleteResult = await library.commitDeletions(deleteIds)
        if case .cancelledByUser = deleteResult {
            phase = .finished
            return deleteResult
        }
    }

    // 2) Album moves. Each group fires one atomic performChanges block
    //    (remove-from-source + add-to-destination), no system prompt.
    for move in moves {
        _ = await library.moveAssets(move.ids, fromAlbumId: move.from, toAlbumId: move.to)
    }

    queue.removeCommitted()
    phase = .finished
    return deleteResult
}

/// `.moveToAlbum` decisions grouped by their (from, to) pair so each atomic
/// PhotoKit call only touches one source-destination combination.
var pendingMoves: [(from: String?, to: String, ids: [AssetIdentifier])] {
    struct Key: Hashable { let from: String?; let to: String }
    var grouped: [Key: [AssetIdentifier]] = [:]
    for d in queue.decisions {
        if case .moveToAlbum(let id, let from, let to) = d {
            grouped[Key(from: from, to: to), default: []].append(id)
        }
    }
    return grouped.map { (from: $0.key.from, to: $0.key.to, ids: $0.value) }
}
```

Add `removeCommitted()` to `DecisionQueue` (replacing `removeDeletions`):

```swift
func removeCommitted() {
    decisions.removeAll { d in
        if case .keep = d { return false }
        return true
    }
}
```
