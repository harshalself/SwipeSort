# SwipeSort v0.5 — Media Polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Prerequisite:** v0.4 tagged. Deletion, persistence, album sort all working.

**Goal:** Make the swipe deck feel real. Add a windowed `PHCachingImageManager` for smooth pre-loading, a 3-slot `AVPlayer` pool for video autoplay, Live Photo hint preview on tap, and pinch-to-zoom for full-resolution inspection. This is the version that separates "prototype" from "usable daily."

**Architecture:** New `AssetPreviewViewModel` per card handles media-type dispatch (photo / video / Live Photo). `PhotoKitLibrary` gains a `WindowedThumbnailCache` for pre-caching. `VideoPlayerPool` is a shared singleton (bounded to 3 players). A `FullResViewer` sheet handles pinch-zoom.

**Tech Stack:** Adds `AVFoundation`, `AVKit`, `PHLivePhotoView` (via `UIViewRepresentable`).

## Global Constraints

- All prior constraints apply.
- **Video plays only for the visible card.** Neighbors are prepared but paused.
- v0.5 autoplays every video regardless of length. v0.7 introduces the ≤30s autoplay threshold + user override; when integrating, expose a `shouldAutoplay(duration:)` decision hook in `VideoCardView` so v0.7 can wire it to `AppSettings` without restructuring.
- **Live Photos never autoplay `.full`** — user must long-press to trigger.
- Cache window: 5 ahead, 2 behind current index. Any change to window must call `stopCachingImages(for:...)` with **identical parameters** to the original start call.
- All `targetSize` values are in pixels (`points × UIScreen.main.scale`). Enforced by helper.

- `PHCachingImageManager` window discipline: `docs/RESEARCH.md` §4.2.
- Video pool of 3 `AVPlayer` (not `AVQueuePlayer`): §4.3.
- Live Photo `.hint` on appear, never autoplay full: §4.4.

```swift
func identifiers(in range: Range<Int>) async -> [AssetIdentifier] {
    let result = await ensureFetchResult()
    let clamped = range.clamped(to: 0..<result.count)
    return clamped.map { AssetIdentifier(result.object(at: $0).localIdentifier) }
}

// The 4-tuple (targetSize, contentMode, options) MUST be identical between
// startCaching* and stopCaching* calls or PHCachingImageManager will not
// release the cached memory. Keep the builder in one place.
private func cachingRequestOptions() -> PHImageRequestOptions {
    let options = PHImageRequestOptions()
    options.deliveryMode = .opportunistic
    options.resizeMode = .fast
    options.isNetworkAccessAllowed = true
    return options
}

func startCachingThumbnails(_ ids: [AssetIdentifier], targetSize: CGSize) {
    // Resolve identifiers directly via fetchAssets(withLocalIdentifiers:) so
    // this works even when the cached PHFetchResult is nil (e.g., right
    // after applyFilter clears it in v0.6) or filtered to a different set.
    let raw = ids.map(\.rawValue)
    let fetched = PHAsset.fetchAssets(withLocalIdentifiers: raw, options: nil)
    var assets: [PHAsset] = []
    fetched.enumerateObjects { asset, _, _ in assets.append(asset) }
    guard !assets.isEmpty else { return }
    imageManager.startCachingImages(
        for: assets,
        targetSize: targetSize,
        contentMode: .aspectFill,
        options: cachingRequestOptions()
    )
}

func stopCachingThumbnails(_ ids: [AssetIdentifier], targetSize: CGSize) {
    let raw = ids.map(\.rawValue)
    let fetched = PHAsset.fetchAssets(withLocalIdentifiers: raw, options: nil)
    var assets: [PHAsset] = []
    fetched.enumerateObjects { asset, _, _ in assets.append(asset) }
    guard !assets.isEmpty else { return }
    imageManager.stopCachingImages(
        for: assets,
        targetSize: targetSize,
        contentMode: .aspectFill,
        options: cachingRequestOptions()
    )
}

func stopCachingAllThumbnails() {
    imageManager.stopCachingImagesForAllAssets()
}
```

- [ ] **Step 3: Implement**

```swift
import Foundation
import UIKit

@MainActor
final class WindowedThumbnailCache {

    private let library: PhotoLibraryProviding
    private let aheadOf: Int
    private let behind: Int
    private let targetSize: CGSize

    private var currentWindow: Range<Int> = 0..<0

    init(library: PhotoLibraryProviding, aheadOf: Int = 5, behind: Int = 2, targetSize: CGSize) {
        self.library = library
        self.aheadOf = aheadOf
        self.behind = behind
        self.targetSize = targetSize
    }

    func slide(to currentIndex: Int, total: Int) async {
        let lower = max(0, currentIndex - behind)
        let upper = min(total, currentIndex + aheadOf + 1)
        let newWindow = lower..<upper

        let newSet = Set(newWindow)
        let oldSet = Set(currentWindow)
        let toStartIndices = newSet.subtracting(oldSet).sorted()
        let toStopIndices = oldSet.subtracting(newSet).sorted()

        // Batch each side into a single call, but resolve identifiers
        // index-by-index so sparse diffs (mid-scroll jumps) are correct.
        if !toStartIndices.isEmpty {
            let ids = await resolveIdentifiers(at: toStartIndices)
            if !ids.isEmpty {
                library.startCachingThumbnails(ids, targetSize: targetSize)
            }
        }
        if !toStopIndices.isEmpty {
            let ids = await resolveIdentifiers(at: toStopIndices)
            if !ids.isEmpty {
                library.stopCachingThumbnails(ids, targetSize: targetSize)
            }
        }

        currentWindow = newWindow
    }

    /// Fetches identifiers for a (possibly sparse) set of indices. Falls
    /// back to per-index resolution because `PhotoLibraryProviding` only
    /// offers a contiguous-range accessor.
    private func resolveIdentifiers(at indices: [Int]) async -> [AssetIdentifier] {
        guard let first = indices.first, let last = indices.last else { return [] }
        let range = await library.identifiers(in: first..<(last + 1))
        // Map absolute index → identifier via offset. The provider is
        // contract-bound to return one identifier per index in the range.
        var out: [AssetIdentifier] = []
        out.reserveCapacity(indices.count)
        for idx in indices {
            let offset = idx - first
            if offset >= 0, offset < range.count {
                out.append(range[offset])
            }
        }
        return out
    }

    func stopAll() {
        library.stopCachingAllThumbnails()
        currentWindow = 0..<0
    }
}
```

Note: `resolveIdentifiers(at:)` still fetches the full contiguous range spanning `min..<max+1` and then indexes into it. For contiguous windows (the common case during forward swipes) that's optimal; for rare mid-scroll jumps it over-fetches at most one window's worth. Acceptable for v0.5.

- [ ] **Step 2: `VideoPlayerPool`**

```swift
import AVFoundation

/// LRU pool of at most 3 `AVPlayer` instances. Least-recently-used eviction
/// (not FIFO) so that undo — which returns to a recently-visible card — is
/// still a cache hit. FIFO would evict the card the user just swiped away
/// from, guaranteeing a re-fetch on undo.
@MainActor
final class VideoPlayerPool {
    static let shared = VideoPlayerPool()

    private struct Slot {
        var assetId: AssetIdentifier
        var player: AVPlayer
        var lastTouched: Date
    }

    private var slots: [Slot] = []
    private let capacity = 3

    private init() {}

    func player(for id: AssetIdentifier, item: AVPlayerItem) -> AVPlayer {
        // Hit: bump `lastTouched` and reuse.
        if let idx = slots.firstIndex(where: { $0.assetId == id }) {
            slots[idx].lastTouched = .now
            return slots[idx].player
        }

        let player = AVPlayer(playerItem: item)
        player.isMuted = true
        player.automaticallyWaitsToMinimizeStalling = false
        item.preferredForwardBufferDuration = 5
        item.preferredPeakBitRate = 2_000_000

        if slots.count >= capacity {
            // Evict LRU.
            if let evictIdx = slots.enumerated()
                .min(by: { $0.element.lastTouched < $1.element.lastTouched })?.offset {
                let evicted = slots.remove(at: evictIdx)
                evicted.player.pause()
                evicted.player.replaceCurrentItem(with: nil)
            }
        }
        slots.append(Slot(assetId: id, player: player, lastTouched: .now))
        return player
    }

    func pauseAllExcept(_ id: AssetIdentifier) {
        for slot in slots where slot.assetId != id {
            slot.player.pause()
        }
    }
}
```

```swift
import SwiftUI
import PhotosUI

/// SwiftUI wrapper for `PHLivePhotoView`. Tap the view to play the short
/// `.hint` preview animation. Never autoplays the full clip — that would
/// force decoders open for every visible card and torch the battery.
struct LivePhotoCardView: UIViewRepresentable {
    let assetId: AssetIdentifier?
    let library: PhotoLibraryProviding
    let fallback: AssetThumbnail?

    final class Coordinator: NSObject {
        weak var livePhotoView: PHLivePhotoView?

        @objc func handleTap() {
            guard let view = livePhotoView, view.livePhoto != nil else { return }
            view.stopPlayback()
            view.startPlayback(with: .hint)
        }
    }

    func makeCoordinator() -> Coordinator { Coordinator() }

    func makeUIView(context: Context) -> PHLivePhotoView {
        let v = PHLivePhotoView()
        v.contentMode = .scaleAspectFill
        v.clipsToBounds = true
        v.layer.cornerRadius = 24
        v.isUserInteractionEnabled = true

        context.coordinator.livePhotoView = v
        let tap = UITapGestureRecognizer(
            target: context.coordinator,
            action: #selector(Coordinator.handleTap)
        )
        v.addGestureRecognizer(tap)
        return v
    }

    func updateUIView(_ uiView: PHLivePhotoView, context: Context) {
        guard let id = assetId else {
            uiView.livePhoto = nil
            return
        }
        // Snapshot layout on the main actor; the request itself is async
        // so we don't block the view update.
        let scale = UIScreen.main.scale
        let target = CGSize(
            width: max(uiView.bounds.width, 1) * scale,
            height: max(uiView.bounds.height, 1) * scale
        )
        Task { @MainActor in
            let live = await library.livePhoto(for: id, targetSize: target)
            uiView.livePhoto = live
            // Do NOT startPlayback here — tap triggers the hint preview.
        }
    }

    static func dismantleUIView(_ uiView: PHLivePhotoView, coordinator: Coordinator) {
        uiView.stopPlayback()
        uiView.livePhoto = nil
    }
}
```

The `SwipeCardView` dispatch already routes `.livePhoto` cases here (see Task 3). No further changes needed in `SwipeCardView` — the tap-to-hint interaction lives inside the coordinator so it doesn't collide with the outer swipe `DragGesture` (`UITapGestureRecognizer` and SwiftUI `DragGesture` cooperate correctly for taps vs drags).

```swift
    @State private var offset: CGSize = .zero

            if let image {

                    .resizable()
                    .scaledToFit()
                    .scaleEffect(scale)
                    .offset(offset)
                    .gesture(
                        SimultaneousGesture(
                            MagnificationGesture()
                                .onChanged { scale = max(1, min(6, $0)) },
                            DragGesture()
                                .onChanged { offset = $0.translation }
                                .onEnded { _ in
                                    if scale <= 1 { withAnimation { offset = .zero } }
                                }
                        )
                    )
                    .onTapGesture(count: 2) {
                        withAnimation { scale = scale > 1 ? 1 : 2; offset = .zero }
                    }
            } else {
                ProgressView().tint(.white)

        .task {
            let scale = UIScreen.main.scale
```
